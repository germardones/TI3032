5. $lookup — Joins en MongoDB

$lookup es la etapa del pipeline que permite unir documentos de dos colecciones, equivalente al JOIN de SQL. MongoDB, siendo una base documental, favorece los datos embebidos; pero cuando las colecciones están separadas y necesitas combinarlas, $lookup es la herramienta.

Sintaxis básica

db.coleccionOrigen.aggregate([
  {
    $lookup: {
      from:         "coleccion_destino",  // colección con la que se une
      localField:   "campo_local",       // campo en el doc actual
      foreignField: "campo_foraneo",     // campo equivalente en destino
      as:           "nombre_resultado"  // nombre del arreglo de resultados
    }
  }
])

El resultado de $lookup es siempre un arreglo de documentos coincidentes, añadido al documento original como un nuevo campo. Si no hay coincidencias, el arreglo queda vacío ([]).

Comparación con SQL JOIN

SQL — LEFT JOIN

-- Une pedidos con clientes
SELECT p.*, c.nombre, c.email
FROM pedidos p
LEFT JOIN clientes c
  ON p.cliente_id = c._id;

MongoDB — $lookup

// Une pedidos con clientes
db.pedidos.aggregate([
  {
    $lookup: {
      from:         "clientes",
      localField:   "cliente_id",
      foreignField: "_id",
      as:           "cliente"
    }
  }
])

Nota: $lookup siempre produce un LEFT OUTER JOIN. Los documentos de la colección origen aparecen siempre en la salida, incluso si no tienen coincidencias en la colección destino (el arreglo resultante estará vacío en ese caso).

Ejemplo 1 — Citas con datos del paciente

Tenemos las colecciones citas y pacientes separadas. Queremos enriquecer cada cita con la información del paciente correspondiente.

// Colección: citas (cada doc tiene un campo paciente_id)
// Colección: pacientes (cada doc tiene _id y datos del paciente)
db.citas.aggregate([
  {
    $lookup: {
      from:         "pacientes",
      localField:   "paciente_id",
      foreignField: "_id",
      as:           "paciente"
    }
  },
  {
    $unwind: "$paciente"            // convierte el arreglo en subdocumento
  },
  {
    $project: {
      _id:             1,
      fecha:           1,
      estado:          1,
      "paciente.nombre": 1,
      "paciente.rut":    1,
      "paciente.edad":   1
    }
  }
])

/* Resultado:
[
  {
    _id: ObjectId("..."),
    fecha: ISODate("2024-03-15"),
    estado: "REALIZADA",
    paciente: { nombre: "Ana Soto", rut: "12.345.678-9", edad: 34 }
  },
  ...
] */

Ejemplo 2 — Pipeline completo: citas con médico y diagnóstico

Combina $match, $lookup, $unwind y $project para generar un reporte completo.

db.citas.aggregate([
  {
    $match: { estado: "REALIZADA", fecha: { $gte: ISODate("2024-01-01") } }
  },
  {
    $lookup: {
      from:         "medicos",
      localField:   "medico_id",
      foreignField: "_id",
      as:           "medico"
    }
  },
  {
    $unwind: "$medico"
  },
  {
    $lookup: {
      from:         "pacientes",
      localField:   "paciente_id",
      foreignField: "_id",
      as:           "paciente"
    }
  },
  {
    $unwind: "$paciente"
  },
  {
    $project: {
      _id:                   0,
      fecha:                 1,
      diagnostico:           1,
      "medico.nombre":       1,
      "medico.especialidad": 1,
      "paciente.nombre":     1,
      "paciente.edad":       1
    }
  }
])

Ejemplo 3 — Contar citas por paciente (con $lookup + $group)

// Objetivo: listado de pacientes con su número de citas realizadas
db.pacientes.aggregate([
  {
    $lookup: {
      from:         "citas",
      localField:   "_id",
      foreignField: "paciente_id",
      as:           "citas"
    }
  },
  {
    $project: {
      _id:          0,
      nombre:       1,
      rut:          1,
      total_citas:  { $size: "$citas" }   // tamaño del arreglo resultado
    }
  },
  {
    $sort: { total_citas: -1 }
  }
])

/* Resultado:
[
  { nombre: "Ana Soto",    rut: "12.345.678-9", total_citas: 8 },
  { nombre: "Carlos Vera", rut: "9.876.543-2",  total_citas: 5 },
  ...
] */

Forma avanzada: $lookup con pipeline

Desde MongoDB 3.6, $lookup acepta un sub-pipeline para aplicar filtros y transformaciones sobre la colección destino antes de unir. Esto permite joins condicionales más precisos.

// Solo traer citas REALIZADAS de cada paciente
db.pacientes.aggregate([
  {
    $lookup: {
      from: "citas",
      let:  { pid: "$_id" },            // variable local para usar en el pipeline
      pipeline: [
        {
          $match: {
            $expr: {
              $and: [
                { $eq: ["$paciente_id", "$$pid"] },   // $$pid referencia la variable local
                { $eq: ["$estado", "REALIZADA"] }
              ]
            }
          }
        },
        { $project: { _id: 0, fecha: 1, diagnostico: 1 } }
      ],
      as: "citas_realizadas"
    }
  }
])

Cuándo usar $lookup vs datos embebidos: si los datos relacionados se consultan juntos con frecuencia, embeberlos es más eficiente. Si las entidades se gestionan por separado o crecen sin límite (ej. historial médico), referencias + $lookup es la opción correcta.
